         ```typescript
         io.on("connection", (socket) => {
           console.log(`[socket.io] Client connected: ${socket.id}`);

           // Handle reconnection via connection state recovery
           if (socket.recovered) {
             const { roomCode } = socket.data;
             if (roomCode) {
               markPlayerReconnected(roomCode, socket.id);
               io.to(roomCode).emit("room:player-reconnected", { socketId: socket.id });
               const state = getRoomStatePayload(roomCode);
               if (state) socket.emit("room:state", state);
             }
           }

           registerRoomHandlers(io, socket);
           registerTurnHandlers(io, socket);
           registerChatHandlers(io, socket);
         });
         ```
